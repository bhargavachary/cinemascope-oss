# Cinemascope

A personal movies & series tracker with two browsing modes:

- **Mosaic View** — visual, poster-first browsing
- **Table View** — compact, sortable list

Run locally with:

1. python3 -m venv venv && source venv/bin/activate
2. pip install -r requirements.txt
3. (optional) python3 scripts/fetch_assets.py
4. OMDB_API_KEY=<your_key> python3 server.py

Live demo: https://bhargavachary.in/movies-series-tracker/

Contributions: PRs welcome for fixes and small improvements. For metadata corrections, edit `data/library.json` or add overrides in `data/overrides.json`.

/* ── 5. Parse data ── */
function parseData(raw) {
    return raw.trim().split('\n').map((line, idx) => {
        // Robust split: try real tab first, fall back to 2+ spaces
        const parts = line.includes('\t') ? line.split('\t') : line.split(/\s{2,}/);
        const id     = parts[0] || String(idx + 1);
        const title  = (parts[1] || '').trim();
        const type   = (parts[2] || 'Movie').trim();
        const imdb   = (parts[3] || '').trim();
        const status = (parts[6] || 'Unknown').trim();

        const rating = (imdb === '—' || imdb === 'N/A' || !imdb) ? 'NR' : imdb;

        /* Deterministic metadata — same index = same values, always */
        const synIdx  = idx % SYNOPSES.length;
        const crtIdx  = idx % CRITICS_BLURBS.length;
        const awdIdx  = idx % AWARDS_POOL.length;

        const yearSeed  = title.split('').reduce((a, c) => a + c.charCodeAt(0), 0);
        const year      = 1985 + (yearSeed % 40);
        const durH      = 1 + (yearSeed % 2);
        const durM      = 5 + (yearSeed % 55);
        const duration  = type === 'Series' ? `${1 + (yearSeed % 5)} seasons` : `${durH}h ${durM}m`;

        return {
            id, title, type, rating, status,
            year,
            duration,
            poster:   getStableImage(title),
            synopsis: SYNOPSES[synIdx],
            critics:  CRITICS_BLURBS[crtIdx],
            awards:   AWARDS_POOL[awdIdx] || [],
        };
    });
}

#!/usr/bin/env python3
"""
Local Admin Server (Flask)
==========================
Serves the static site locally and exposes admin API endpoints for
metadata refresh and library edits.

Usage:
    OMDB_API_KEY=your_key python3 server.py
    Then open http://localhost:8080
"""

import json
import os
from pathlib import Path
from difflib import SequenceMatcher

import requests
from flask import Flask, jsonify, request, send_from_directory

BASE_DIR = Path(__file__).parent.resolve()
DATA_FILE = BASE_DIR / "data" / "library.json"

OMDB_API_KEY = os.environ.get("OMDB_API_KEY", "")

WIKIDATA_SEARCH = "https://www.wikidata.org/w/api.php"
WIKIDATA_ENTITY = "https://www.wikidata.org/wiki/Special:EntityData/{qid}.json"
WIKIPEDIA_SUMMARY = "https://en.wikipedia.org/api/rest_v1/page/summary/{title}"

app = Flask(__name__, static_folder=None)


def load_library():
    with open(DATA_FILE, "r", encoding="utf-8") as f:
        return json.load(f)


def save_library(data):
    with open(DATA_FILE, "w", encoding="utf-8") as f:
        json.dump(data, f, indent=4, ensure_ascii=False)
        f.write("\n")


def normalize_title(value):
    return "".join(ch for ch in value.lower() if ch.isalnum())


def _omdb_get(params):
    if not OMDB_API_KEY:
        return None
    try:
        resp = requests.get("https://www.omdbapi.com/", params=params, timeout=10)
        resp.raise_for_status()
        data = resp.json()
        if data.get("Response") == "True":
            return data
    except Exception:
        return None
    return None


def fetch_omdb_by_id(imdb_id):
    if not imdb_id:
        return None
    return _omdb_get({"apikey": OMDB_API_KEY, "i": imdb_id, "plot": "short"})


def fetch_omdb_by_title(title, media_type):
    params = {"apikey": OMDB_API_KEY, "t": title, "plot": "short"}
    if media_type:
        params["type"] = media_type
    data = _omdb_get(params)
    if data:
        return data

    try:
        params = {"apikey": OMDB_API_KEY, "s": title}
        if media_type:
            params["type"] = media_type
        resp = requests.get("https://www.omdbapi.com/", params=params, timeout=10)
        resp.raise_for_status()
        payload = resp.json()
        if payload.get("Response") != "True":
            return None
        candidates = payload.get("Search", [])
        if not candidates:
            return None
        target = normalize_title(title)
        best = None
        best_score = 0
        for c in candidates:
            score = SequenceMatcher(None, target, normalize_title(c.get("Title", ""))).ratio()
            if score > best_score:
                best_score = score
                best = c
        if best and best_score >= 0.72:
            return fetch_omdb_by_id(best.get("imdbID"))
    except Exception:
        return None
    return None


def wikidata_search(title, media_type):
    try:
        params = {
            "action": "wbsearchentities",
            "search": title,
            "language": "en",
            "format": "json",
            "limit": 5,
        }
        r = requests.get(WIKIDATA_SEARCH, params=params, timeout=10)
        r.raise_for_status()
        results = r.json().get("search", [])
        if not results:
            return None

        for item in results:
            desc = (item.get("description") or "").lower()
            if media_type == "movie" and "film" in desc:
                return item.get("id")
            if media_type == "series" and ("television" in desc or "tv" in desc or "series" in desc):
                return item.get("id")

        return results[0].get("id")
    except Exception:
        return None


def wikidata_entity(qid):
    try:
        r = requests.get(WIKIDATA_ENTITY.format(qid=qid), timeout=10)
        r.raise_for_status()
        return r.json()["entities"][qid]
    except Exception:
        return None


def entity_claim(claims, pid):
    claim = claims.get(pid, [])
    if not claim:
        return None
    mainsnak = claim[0].get("mainsnak", {})
    datavalue = mainsnak.get("datavalue", {})
    return datavalue.get("value")


def wikipedia_summary(title):
    try:
        r = requests.get(WIKIPEDIA_SUMMARY.format(title=title.replace(" ", "%20")), timeout=10)
        if r.status_code != 200:
            return None
        return r.json()
    except Exception:
        return None


def fetch_wikidata_fallback(title, media_type):
    qid = wikidata_search(title, media_type)
    if not qid:
        return {}
    entity = wikidata_entity(qid)
    if not entity:
        return {}

    claims = entity.get("claims", {})
    imdb_id = entity_claim(claims, "P345")
    runtime = entity_claim(claims, "P2047")
    sitelinks = entity.get("sitelinks", {})
    wikipedia_title = sitelinks.get("enwiki", {}).get("title")

    meta = {}
    if imdb_id:
        meta["imdb_id"] = imdb_id
    if wikipedia_title:
        meta["wikipedia"] = f"https://en.wikipedia.org/wiki/{wikipedia_title.replace(' ', '_')}"
        summary = wikipedia_summary(wikipedia_title)
        if summary and summary.get("extract"):
            meta["plot"] = summary.get("extract")
    if runtime and isinstance(runtime, dict):
        amount = runtime.get("amount")
        if amount:
            meta["runtime"] = f"{amount.replace('+', '')} min"

    return meta


def refresh_metadata(item):
    title = item.get("title", "")
    media_type = "movie" if item.get("type", "Movie") == "Movie" else "series"

    omdb = None
    if OMDB_API_KEY:
        if item.get("imdbID"):
            omdb = fetch_omdb_by_id(item.get("imdbID"))
        if not omdb:
            omdb = fetch_omdb_by_title(title, media_type)

    ratings = {}
    if omdb:
        ratings = {r.get("Source"): r.get("Value") for r in omdb.get("Ratings", [])}

    meta = {
        "imdbID": omdb.get("imdbID") if omdb else item.get("imdbID"),
        "imdb_rating": omdb.get("imdbRating") if omdb else "",
        "metascore": omdb.get("Metascore") if omdb else "",
        "rotten_tomatoes": ratings.get("Rotten Tomatoes", ""),
        "plot": omdb.get("Plot") if omdb else "",
        "runtime": omdb.get("Runtime") if omdb else "",
    }

    # Fallback/Enhancement via Wikidata (The "custom script" logic)
    # We try this if we are missing data OR if the user asked for a refresh (which is this function).
    # We prioritize Wikidata for Plot if OMDB's is short or missing.
    wiki_meta = fetch_wikidata_fallback(title, media_type)
    
    if wiki_meta:
        # Fill missing fields
        for key, value in wiki_meta.items():
            if not meta.get(key):
                meta[key] = value
        
        # specific overwrite logic
        if wiki_meta.get("plot") and len(wiki_meta.get("plot")) > len(meta.get("plot", "")):
            meta["plot"] = wiki_meta["plot"]
        if wiki_meta.get("wikipedia"):
             meta["wikipedia"] = wiki_meta["wikipedia"]

    meta["imdb_id"] = meta.get("imdbID") or meta.get("imdb_id")

    item["imdbID"] = meta.get("imdbID") or item.get("imdbID")
    item["plot"] = meta.get("plot") or item.get("plot")
    item["runtime"] = meta.get("runtime") or item.get("runtime")
    item["imdbRating"] = meta.get("imdb_rating") or item.get("imdbRating")
    item["metascore"] = meta.get("metascore") or item.get("metascore")
    item["rottenTomatoes"] = meta.get("rotten_tomatoes") or item.get("rottenTomatoes")
    item["wikipedia"] = meta.get("wikipedia") or item.get("wikipedia")

    return meta

@app.route("/api/library", methods=["POST"])
def api_add_item():
    library = load_library()
    body = request.json or {}
    max_id = max((item["id"] for item in library), default=0)
    body["id"] = max_id + 1
    if not body.get("title"):
        return jsonify({"error": "Title is required"}), 400
    item = {
        "id": body["id"],
        "title": body["title"],
        "type": body.get("type", "Movie"),
        "status": body.get("status", "Plan to Watch"),
        "language": body.get("language", "English"),
        "rating": body.get("rating", 0),
        "genres": body.get("genres", []),
        "imdbID": body.get("imdbID"),
        "tags": body.get("tags", [])
    }
    library.append(item)
    save_library(library)
    return jsonify(item), 201


@app.route("/api/library/<int:item_id>", methods=["PUT"])
def api_update_item(item_id):
    library = load_library()
    body = request.json or {}
    for i, item in enumerate(library):
        if item["id"] == item_id:
            library[i] = {
                **item,
                "title": body.get("title", item["title"]),
                "type": body.get("type", item["type"]),
                "status": body.get("status", item["status"]),
                "language": body.get("language", item.get("language", "English")),
                "rating": body.get("rating", item.get("rating", 0)),
                "genres": body.get("genres", item.get("genres", [])),
                "imdbID": body.get("imdbID", item.get("imdbID")),
                "tags": body.get("tags", item.get("tags", []))
            }
            save_library(library)
            return jsonify(library[i])
    return jsonify({"error": "Item not found"}), 404


@app.route("/api/library/<int:item_id>", methods=["DELETE"])
def api_delete_item(item_id):
    library = load_library()
    new_library = [item for item in library if item["id"] != item_id]
    if len(new_library) == len(library):
        return jsonify({"error": "Item not found"}), 404
    save_library(new_library)
    return jsonify({"deleted": item_id})


@app.route("/api/refresh_metadata/<int:item_id>", methods=["POST"])

/**
 * Cinemascope — Application Logic
 * ─────────────────────────────────
 * Unified controller for Table + Mosaic views.
 * Loads data from assets/manifest.json, applies local overrides,
 * and renders to whichever view is active.
 *
 * Design principles:
 *   • Single source of truth for state
 *   • Pure render functions (state → DOM)
 *   • Debounced search
 *   • Stagger animations via CSS animation-delay
 *   • Keyboard-accessible modals
 */
(function () {
    'use strict';

    /* ═══════════════════════════════════════════════════
       STATE
       ═══════════════════════════════════════════════════ */
    let allItems = [];
    let activeStatus = 'all';
    let activeType   = 'all';
    let activeSort   = 'imdb_rating';
    let searchQuery  = '';
    let currentView  = localStorage.getItem('cinemascope-view') || 'table';
    let currentItem  = null;
    const isLocal    = ['localhost', '127.0.0.1', '0.0.0.0'].includes(location.hostname);

    /* ═══════════════════════════════════════════════════
       DOM REFS  (resolved once on boot)
       ═══════════════════════════════════════════════════ */
    const $ = (s) => document.querySelector(s);
    const $$ = (s) => document.querySelectorAll(s);

    const dom = {
        statsSummary:  $('#stats-summary'),
        viewToggle:    $('#view-toggle'),
        viewIcon:      $('#view-toggle-icon'),
        viewLabel:     $('#view-toggle-label'),
        sortSelect:    $('#sort-select'),
        searchInput:   $('#search-input'),
        resultsCount:  $('#results-count'),
        tableView:     $('#table-view'),
        tableBody:     $('#table-body'),
        mosaicView:    $('#mosaic-view'),
        emptyState:    $('#empty-state'),
        modalBackdrop: $('#modal-backdrop'),
        modalClose:    $('#modal-close'),
        modalPoster:   $('#modal-poster'),
        modalTitle:    $('#modal-title'),
        modalSubtitle: $('#modal-subtitle'),
        modalMeta:     $('#modal-meta'),
        modalPlot:     $('#modal-plot'),
        modalLinks:    $('#modal-links'),
        adminRefresh:  $('#admin-refresh'),
    };

    /* ═══════════════════════════════════════════════════
       BOOT
       ═══════════════════════════════════════════════════ */
    document.addEventListener('DOMContentLoaded', boot);

    async function boot() {
        applyView(currentView);
        bindEvents();
        try {
            const resp = await fetch('assets/manifest.json');
            if (!resp.ok) throw new Error('Manifest not found');
            const raw = await resp.json();
            const checked = await checkPosters(raw);
            allItems = applyOverrides(checked);
            renderStats();
            render();
        } catch (err) {
            console.error('Boot error:', err);
            showEmpty();
        }
    }

    /* ═══════════════════════════════════════════════════
       DATA HELPERS
       ═══════════════════════════════════════════════════ */
    function getOverrides() {
        try { return JSON.parse(localStorage.getItem('metadata_overrides_v1') || '{}'); }
        catch { return {}; }
    }

    function applyOverrides(items) {
        const ov = getOverrides();
        return items.map(item => {
            const o = ov[item.title];
            return o ? Object.assign({}, item, o) : item;
        });
    }

    async function checkPosters(items) {
        const results = await Promise.allSettled(
            items.map(item => {
                const url = 'assets/posters/' + item.src;
                return fetch(url, { method: 'HEAD' })
                    .then(r => { item.has_poster = r.ok; return item; })
                    .catch(() => { item.has_poster = false; return item; });
            })
        );
        return results.map(r => r.value || r.reason);
    }

    /* ═══════════════════════════════════════════════════
       FILTER + SORT
       ═══════════════════════════════════════════════════ */
    function getFiltered() {
        return allItems.filter(item => {
            if (activeStatus !== 'all' && item.status !== activeStatus) return false;
            if (activeType !== 'all' && item.type !== activeType) return false;
            if (searchQuery && !item.title.toLowerCase().includes(searchQuery)) return false;
            return true;
        });
    }

    function applySorted(items) {
        const list = items.slice();
        const num = v => {
            if (!v) return -1;
            if (typeof v === 'number') return v;
            const s = String(v).replace('%', '');
            return parseFloat(s) || -1;
        };
        switch (activeSort) {
            case 'title':
                list.sort((a, b) => a.title.localeCompare(b.title));
                break;
            case 'imdb_rating':
                list.sort((a, b) => num(b.imdb_rating) - num(a.imdb_rating));
                break;
            case 'metascore':
                list.sort((a, b) => num(b.metascore) - num(a.metascore));
                break;
            case 'rotten_tomatoes':
                list.sort((a, b) => num(b.rotten_tomatoes) - num(a.rotten_tomatoes));
                break;
            case 'status':
                list.sort((a, b) => a.status.localeCompare(b.status));
                break;
        }
        // Push items without posters to end
        const withPoster = list.filter(i => i.has_poster);
        const noPoster   = list.filter(i => !i.has_poster);
        return withPoster.concat(noPoster);
    }

    /* ═══════════════════════════════════════════════════
       RENDER ORCHESTRATOR
       ═══════════════════════════════════════════════════ */
    function render() {
        const filtered = applySorted(getFiltered());
        dom.resultsCount.textContent = filtered.length + ' shown';

        if (filtered.length === 0) {
            showEmpty();
            return;
        }
        hideEmpty();

        if (currentView === 'table') {
            renderTable(filtered);
        } else {
            renderMosaic(filtered);
        }
    }

    function showEmpty() {
        dom.tableView.style.display = 'none';
        dom.mosaicView.style.display = 'none';
        dom.emptyState.style.display = 'block';
    }

    function hideEmpty() {
        dom.emptyState.style.display = 'none';
        if (currentView === 'table') {
            dom.tableView.style.display = '';
            dom.mosaicView.style.display = 'none';
        } else {
            dom.tableView.style.display = 'none';
            dom.mosaicView.style.display = '';
        }
    }

    /* ═══════════════════════════════════════════════════
       TABLE RENDERER
       ═══════════════════════════════════════════════════ */
    function renderTable(items) {
        const frag = document.createDocumentFragment();

        items.forEach((item, idx) => {
            const tr = document.createElement('tr');

            // Rating color class
            const imdbVal = parseFloat(item.imdb_rating) || 0;
            const ratingClass = imdbVal >= 8 ? 'rating-high' : imdbVal >= 6 ? 'rating-mid' : 'rating-low';

            // Status class
            const statusKey = item.status === 'Watched' ? 's-watched'
                            : item.status === 'Recommended' ? 's-recommended'
                            : 's-planned';

            const imdbLink = item.imdb_id
                ? '<a href="https://www.imdb.com/title/' + esc(item.imdb_id) + '" target="_blank" rel="noopener">IMDb↗</a>'

#!/usr/bin/env python3
"""
Enhance library.json with new fields: imdbID, language, genres

import json

def infer_language(title, tags):
    """Infer language/region from title and tags"""
    tags_lower = [t.lower() for t in tags]
    
    # Check explicit language tags
    if 'malayalam' in tags_lower:
        return 'Malayalam'
    if 'tamil' in tags_lower:
        return 'Tamil'
    if 'hindi' in tags_lower:
        return 'Hindi'
    if 'telugu' in tags_lower:
        return 'Telugu'
    if 'odia' in ' '.join(tags_lower):
        return 'Odia'
    if 'kannada' in tags_lower:
        return 'Kannada'
    
    # Check title for Indian language patterns
    regional_language_titles = [
        'Kantara', 'Bison', 'Kaantha', 'Dhurandhar', 'Maranamass', 
        'Pravinkoodu Shappu', 'Kishkindha Kaandam', 'Bramayugam',
        'Maharaja', 'Drishyam', '12th Fail', '3 Idiots', 'Taare Zameen Par',
        'Daman', 'Madharaasi', 'Retro', 'Sector 36', 'Article 15',
        'Kohrra', 'Dilli Dark', 'Stolen', 'Nishaanchi', 'Haq',
        'Homebound', 'Lokah', 'Hrudayapoorvam', 'Jugnuma', 'Ponman',
        'Sarvam Maya', 'Aan Paavam Pollathathu', 'Sister Midnight',
        'Thalavara', 'Bou Buttu Bhuta', 'Ronth'
    ]
    for regional_title in regional_language_titles:
        if regional_title.lower() in title.lower():
            # Default to Hindi for generic Indian titles
            return 'Hindi'
    
    # Japanese/Anime
    if 'anime' in tags_lower or title in ['Mirai', 'Spirited Away', 'Shōgun']:
        return 'Japanese'
    
    # Default to English for international titles
    return 'English'

def extract_genres(tags):
    """Extract genre tags (max 3)"""
    genre_keywords = [
        'Action', 'Thriller', 'Drama', 'Comedy', 'Sci-Fi', 'Horror',
        'Mystery', 'Crime', 'Fantasy', 'Romance', 'Animation', 'Noir',
        'Superhero', 'Adventure', 'Epic', 'Political', 'Folk Horror',
        'Dark Comedy', 'Time Travel', 'Spy'
    ]
    
    genres = []
    for tag in tags:
        if any(genre.lower() in tag.lower() for genre in genre_keywords):
            genres.append(tag)
        elif tag in genre_keywords:
            genres.append(tag)
    
    # Limit to 3 genres
    return genres[:3]

def enhance_library():
    """Add new fields to library.json"""
    with open('data/library.json', 'r', encoding='utf-8') as f:
        library = json.load(f)
    
    for item in library:
        # Add imdbID field (initially null, will be populated via API later)
        if 'imdbID' not in item:
            item['imdbID'] = None
        
        # Infer language if not present
        if 'language' not in item:
            item['language'] = infer_language(item['title'], item.get('tags', []))
        
        # Extract genres from tags
        if 'genres' not in item:
            genres = extract_genres(item.get('tags', []))
            # If no genre found, try to infer from tags
            if not genres and item.get('tags'):
                # Use first 3 non-language tags as genres
                non_lang_tags = [t for t in item['tags'] if t.lower() not in 
                                ['malayalam', 'tamil', 'hindi', 'telugu', 'odia', 'kannada']]
                genres = non_lang_tags[:3]
            item['genres'] = genres
    
    # Write back to file
    with open('data/library.json', 'w', encoding='utf-8') as f:
        json.dump(library, f, indent=4, ensure_ascii=False)
    
    print(f"Enhanced {len(library)} entries")
    print("\nSample enhanced entries:")
    for item in library[:5]:
        print(f"  {item['title']}: language={item['language']}, genres={item['genres']}")

if __name__ == '__main__':
    enhance_library()

def fetch_omdb(title, media_type=None):
    """Fetch metadata from OMDB with multi-strategy fallback."""
    if not OMDB_API_KEY:
        return None

    base = {"apikey": OMDB_API_KEY, "plot": "short"}

    # Strategy 1: exact title + type
    params = {**base, "t": title}
    if media_type:
        params["type"] = media_type
    data = _omdb_get(params)
    if data:
        return data

    # Strategy 2: exact title, no type constraint
    if media_type:
        data = _omdb_get({**base, "t": title})
        if data:
            return data

    # Strategy 3: strip subtitle after colon
    if ":" in title:
        shorter = title.split(":")[0].strip()
        data = _omdb_get({**base, "t": shorter})
        if data:
            return data

    # Strategy 4: search endpoint + best match
    result = search_omdb(title, media_type)
    if result:
        return result

    return None


def _omdb_get(params):
    """Single OMDB request. Returns dict or None."""
    try:
        resp = requests.get("https://www.omdbapi.com/", params=params, timeout=10)
        resp.raise_for_status()
        data = resp.json()
        if data.get("Response") == "True":
            return data
    except Exception:
        pass
    return None


def wikidata_search(title, media_type):
    try:
        params = {
            "action": "wbsearchentities",
            "search": title,
            "language": "en",
            "format": "json",
            "limit": 5,
        }
        r = requests.get(WIKIDATA_SEARCH, params=params, timeout=10)
        r.raise_for_status()
        results = r.json().get("search", [])
        if not results:
            return None

        for item in results:
            desc = (item.get("description") or "").lower()
            if media_type == "Movie" and "film" in desc:
                return item.get("id")
            if media_type == "Series" and ("television" in desc or "tv" in desc or "series" in desc):
                return item.get("id")

        return results[0].get("id")
    except Exception:
        return None


def wikidata_entity(qid):

def fetch_wikidata_fallback(title, media_type):
    qid = wikidata_search(title, media_type)
    if not qid:

    entity = wikidata_entity(qid)
    if not entity:

def search_omdb(title, media_type=None):
    """Use OMDB search endpoint and pick best fuzzy match by title."""
    try:
        params = {"apikey": OMDB_API_KEY, "s": title}
        if media_type:
            params["type"] = media_type
        resp = requests.get("https://www.omdbapi.com/", params=params, timeout=10)
        resp.raise_for_status()
        data = resp.json()
        if data.get("Response") != "True":
            return None

        candidates = data.get("Search", [])
        if not candidates:
            return None

        target = normalize_title(title)
        best = None
        best_score = 0
        for c in candidates:
            score = SequenceMatcher(None, target, normalize_title(c.get("Title", ""))).ratio()
            if score > best_score:
                best_score = score
                best = c

        if best and best_score >= 0.72:
            imdb_id = best.get("imdbID")
            if imdb_id:

def download_poster(url, save_path):
    """Download a poster, resize to target width, save as optimized JPEG."""
    try:
        resp = requests.get(url, timeout=15)
        resp.raise_for_status()
        img = Image.open(BytesIO(resp.content))
        if img.mode != "RGB":
            img = img.convert("RGB")
        ratio = TARGET_WIDTH / img.width
        new_h = int(img.height * ratio)
        img = img.resize((TARGET_WIDTH, new_h), Image.Resampling.LANCZOS)
        img.save(save_path, "JPEG", quality=85, optimize=True)
        return img.width, img.height
    except Exception as e:
        print(f"    ⚠ Download failed: {e}")
        return None


def generate_placeholder(title, media_type, save_path):
    """Generate a styled placeholder poster card using Pillow."""
    w, h = TARGET_WIDTH, TARGET_HEIGHT
    idx = sum(ord(c) for c in title) % len(PALETTE)
    bg = PALETTE[idx]
    accent = ACCENTS[idx % len(ACCENTS)]

    img = Image.new("RGB", (w, h), bg)
    draw = ImageDraw.Draw(img)
    font_lg, font_sm = _load_fonts()

    # Accent bar at top
    draw.rectangle([0, 0, w, 4], fill=accent)

    # Type badge (upper‑right corner)
    badge = "FILM" if media_type == "Movie" else "SERIES"
    bb = draw.textbbox((0, 0), badge, font=font_sm)
    bw, bh = bb[2] - bb[0], bb[3] - bb[1]
    bx, by = w - bw - 16, 16
    draw.rounded_rectangle(
        [bx - 6, by - 4, bx + bw + 6, by + bh + 4],
        radius=4, fill=accent,
    )
    draw.text((bx, by), badge, fill="#ffffff", font=font_sm)

    # Centered title
    wrapped = textwrap.wrap(title, width=16)
    line_h = 30
    y = (h - len(wrapped) * line_h) // 2
    for line in wrapped:
        lb = draw.textbbox((0, 0), line, font=font_lg)
        lw = lb[2] - lb[0]
        x = (w - lw) // 2
        draw.text((x + 1, y + 1), line, fill="#000000", font=font_lg)
        draw.text((x, y), line, fill="#ffffff", font=font_lg)
        y += line_h

    img.save(save_path, "JPEG", quality=85, optimize=True)
    return w, h

#!/usr/bin/env python3
"""
Backup Metadata Resolver (No Scraping)
======================================
Uses Wikidata + Wikipedia APIs to fill gaps when OMDB mismatches.
Writes results into data/overrides.json (merged by title).

Usage:
    python3 scripts/fetch_metadata_backup.py
    python3 scripts/fetch_metadata_backup.py --limit 10
"""

import argparse
import json
import time
from pathlib import Path
import requests

SCRIPT_DIR = Path(__file__).parent.resolve()
PROJECT_ROOT = SCRIPT_DIR.parent if SCRIPT_DIR.name == "scripts" else SCRIPT_DIR
LIBRARY_PATH = PROJECT_ROOT / "data" / "library.json"
OVERRIDES_PATH = PROJECT_ROOT / "data" / "overrides.json"

WIKIDATA_SEARCH = "https://www.wikidata.org/w/api.php"
WIKIDATA_ENTITY = "https://www.wikidata.org/wiki/Special:EntityData/{qid}.json"
WIKIPEDIA_SUMMARY = "https://en.wikipedia.org/api/rest_v1/page/summary/{title}"

REQUEST_DELAY = 0.2


def parse_args():
    parser = argparse.ArgumentParser(description="Resolve metadata via Wikidata/Wikipedia")
    parser.add_argument("--limit", type=int, help="Process only the first N entries")
    return parser.parse_args()


def load_library():
    with open(LIBRARY_PATH, "r", encoding="utf-8") as f:
        return json.load(f)


def load_overrides():
    if not OVERRIDES_PATH.exists():
        return {}
    with open(OVERRIDES_PATH, "r", encoding="utf-8") as f:
        data = json.load(f)
    return data if isinstance(data, dict) else {}


def save_overrides(data):
    with open(OVERRIDES_PATH, "w", encoding="utf-8") as f:
        json.dump(data, f, indent=2, ensure_ascii=False)


def wikidata_search(title, media_type):
    params = {
        "action": "wbsearchentities",
        "search": title,
        "language": "en",
        "format": "json",
        "limit": 5,
    }
    r = requests.get(WIKIDATA_SEARCH, params=params, timeout=10)
    r.raise_for_status()
    results = r.json().get("search", [])
    if not results:
        return None

    # Prefer items tagged as film or television series when possible
    for item in results:
        desc = (item.get("description") or "").lower()
        if media_type == "Movie" and "film" in desc:
            return item.get("id")
        if media_type == "Series" and ("television" in desc or "tv" in desc or "series" in desc):
            return item.get("id")

    return results[0].get("id")


def entity_claim(claims, pid):
    claim = claims.get(pid, [])
    if not claim:
        return None
    mainsnak = claim[0].get("mainsnak", {})
    datavalue = mainsnak.get("datavalue", {})
    return datavalue.get("value")


def wikidata_entity(qid):
    r = requests.get(WIKIDATA_ENTITY.format(qid=qid), timeout=10)
    r.raise_for_status()
    return r.json()["entities"][qid]


def wikipedia_summary(title):
    r = requests.get(WIKIPEDIA_SUMMARY.format(title=title.replace(" ", "%20")), timeout=10)
    if r.status_code != 200:
        return None
    return r.json()


def extract_metadata(entity):
    claims = entity.get("claims", {})
    imdb_id = entity_claim(claims, "P345")
    image = entity_claim(claims, "P18")
    runtime = entity_claim(claims, "P2047")
    wikipedia_title = None

    sitelinks = entity.get("sitelinks", {})
    if "enwiki" in sitelinks:
        wikipedia_title = sitelinks["enwiki"].get("title")

    meta = {}
    if imdb_id:
        meta["imdb_id"] = imdb_id
    if wikipedia_title:
        meta["wikipedia"] = f"https://en.wikipedia.org/wiki/{wikipedia_title.replace(' ', '_')}"

    if runtime and isinstance(runtime, dict):
        amount = runtime.get("amount")
        if amount:
            meta["runtime"] = f"{amount.replace('+','')} min"

    if image:
        meta["wikidata_image"] = image

    if wikipedia_title:
        summary = wikipedia_summary(wikipedia_title)
        if summary and summary.get("extract"):
            meta["plot"] = summary.get("extract")

    return meta


def main():
    args = parse_args()
    library = load_library()
    if args.limit:
        library = library[: args.limit]

    overrides = load_overrides()

    for entry in library:
        title = entry["title"]
        media_type = entry.get("type", "Movie")

        if title in overrides:
            continue

        try:
            qid = wikidata_search(title, media_type)
            if not qid:
                continue
            entity = wikidata_entity(qid)
            meta = extract_metadata(entity)
            if meta:
                overrides[title] = meta
                print(f"✓ {title} → {qid} ({len(meta)} fields)")
        except Exception:
            pass

        time.sleep(REQUEST_DELAY)

    save_overrides(overrides)
    print(f"\nSaved overrides to {OVERRIDES_PATH}")


if __name__ == "__main__":
    main()

def setup_pillow():
    """Ensure Pillow is installed for image processing"""

        from PIL import Image, ImageOps
        return Image, ImageOps
    except ImportError:
        print("❌ Pillow (PIL) library not found.")

def optim_image(image_path, output_path, max_width=600):
    """Resize and compress image to WebP"""
    try:
        with Image.open(image_path) as img:
            # Convert to RGB (in case of RGBA/P)
            if img.mode in ('RGBA', 'P'):
                img = img.convert('RGB')
            
            # Calculate new height to maintain aspect ratio
            w_percent = (max_width / float(img.size[0]))
            if w_percent < 1: # Only downscale
                h_size = int((float(img.size[1]) * float(w_percent)))
                img = img.resize((max_width, h_size), Image.Resampling.LANCZOS)
            
            # Save as optimized WebP
            img.save(output_path, 'WEBP', quality=80, optimize=True)
            return True
    except Exception as e:
        print(f"   ⚠️ Error processing image {image_path}: {e}")
        return False

def fetch_poster(item):
    """Fetch poster for a single library item"""
    item_id = item.get('id')
    title = item.get('title')
    imdb_id = item.get('imdbID')
    
    # Target filename
    target_filename = f"{item_id}.webp"
    target_path = ASSETS_DIR / target_filename
    
    if target_path.exists():
        print(f"   ✅ Exists: {title}")
        return True

    poster_url = None
    
    # Strategy 1: Use OMDB if API Key exists and we have an IMDb ID or Title
    if OMDB_API_KEY:
        try:
            query = f"i={imdb_id}" if imdb_id else f"t={urllib.parse.quote(title)}"
            url = f"https://www.omdbapi.com/?apikey={OMDB_API_KEY}&{query}"
            
            with urllib.request.urlopen(url) as resp:
                data = json.load(resp)
                if data.get('Response') == 'True' and data.get('Poster') != 'N/A':
                    poster_url = data.get('Poster')
        except Exception as e:
            print(f"   ⚠️ OMDB Fetch Error for {title}: {e}")

    # Strategy 2: Fallback (Not implemented to avoid pure scraping without consent, 
    # but this is where one might add other sources)
    
    if not poster_url:
        print(f"   ⏭️  Skipped (No URL): {title}")
        return False

    # Download and Optimize
    print(f"   ⬇️  Downloading: {title}...")
    try:
        temp_path = ASSETS_DIR / f"temp_{item_id}.jpg"
        
        # Download
        req = urllib.request.Request(poster_url, headers={'User-Agent': 'Mozilla/5.0'})
        with urllib.request.urlopen(req) as response, open(temp_path, 'wb') as out_file:
            out_file.write(response.read())
            
        # Optimize
        success = optim_image(temp_path, target_path)
        
        # Cleanup temp
        if temp_path.exists():
            os.remove(temp_path)
            
        if success:
            print(f"   ✨ Saved: {target_filename}")
            return True
            
    except Exception as e:
        print(f"   ❌ Download Failed for {title}: {e}")
        return False

def main():
    print(f"🎬 Movie Tracker Asset Pipeline")
    print(f"   Target Dir: {ASSETS_DIR}")
    print(f"   Data Source: {DATA_FILE}")

            count += 1
            # Be nice to APIs
            time.sleep(0.1)
            
    print("-" * 40)
    print(f"Done. Processed {len(library)} items. {count} images available locally.")

if __name__ == "__main__":
    main()

        item['external_ratings']['letterboxd'] = letterboxd
    
    imdb = fetch_imdb_rating(item['title'], content_type)
    if imdb:
        item['external_ratings']['imdb'] = imdb
    
    cinemascore = fetch_cinemascore_grade(item['title'])
    if cinemascore and item['type'] == "Movie":  # CinemaScore is for theatrical releases
        item['external_ratings']['cinemascore'] = cinemascore
    
    movielens = fetch_movielens_rating(item['title'])
    if movielens:
        item['external_ratings']['movielens'] = movielens
    
    # Update main rating field if external ratings are available
    # Only update if there's no existing rating (0 is considered 'no rating')
    # Priority: IMDb > Metacritic (converted to /10) > Letterboxd (converted to /10)
    if item['rating'] == 0:
        if imdb and 'rating' in imdb:
            item['rating'] = round(imdb['rating'], 1)
        elif metacritic and 'metascore' in metacritic:
            # Convert Metacritic's 0-100 scale to 0-10 scale
            item['rating'] = round(metacritic['metascore'] / 10, 1)
        elif letterboxd and 'rating' in letterboxd:
            # Convert Letterboxd's 0-5 scale to 0-10 scale
            item['rating'] = round((letterboxd['rating'] / letterboxd['scale']) * 10, 1)
    
    # Rate limiting to be polite to servers
    time.sleep(1)
    
    return item

class UIValidator:
    """Validates UI implementation against design system specifications"""
    
    DESIGN_SYSTEM = {
        "spacing": {
            "8pt_grid": [0.5, 1, 1.5, 2, 3, 4],  # rem values
            "card_radius": [18, 20],  # px
            "control_radius": [12],  # px
        },
        "colors": {
            "sf_black": "#050505",
            "sf_gray_900": "#0B0B0C",

        },
        "motion": {
            "spring": "cubic-bezier(0.16, 1, 0.3, 1)",
            "duration_micro": "0.18s",
            "duration_standard": "0.24s",
        }
    }
    
    def __init__(self, project_root):
        self.project_root = Path(project_root)
        self.css_files = [
            self.project_root / "css" / "styles.css",
            self.project_root / "css" / "styles.addon.css"
        ]
        self.html_file = self.project_root / "index.html"
        self.js_file = self.project_root / "js" / "app.js"
        self.results = []
        
    def validate_all(self):
        """Run all validation checks"""
        print("🔍 Starting UI Validation...")
        print("=" * 60)
        
        self.validate_spacing()
        self.validate_radii()
        self.validate_touch_targets()
        self.validate_motion()
        self.validate_colors()
        self.validate_structure()
        
        self.print_results()
        
    def validate_spacing(self):
        """Validate 8pt grid system compliance"""
        print("\n📏 Validating 8pt Grid System...")
        
        violations = []
        for css_file in self.css_files:
            if not css_file.exists():
                continue
                
            with open(css_file, 'r') as f:
                content = f.read()
                
            # Check for hardcoded pixel values that don't follow 8pt grid
            import re
            # Look for px values that aren't multiples of 8
            px_values = re.findall(r':\s*(\d+)px', content)
            
            for val in px_values:
                num = int(val)
                # Allow certain exceptions (1px borders, etc)
                if num > 4 and num % 8 != 0:
                    violations.append(f"Non-8pt value: {num}px in {css_file.name}")
        
        if violations:
            self.results.append({
                "test": "8pt Grid System",
                "status": "⚠️  WARNINGS",
                "details": violations[:5]  # Limit to first 5
            })
        else:
            self.results.append({
                "test": "8pt Grid System",
                "status": "✅ PASSED",
                "details": []
            })
    
    def validate_radii(self):
        """Validate border-radius values match squircle standards"""
        print("📐 Validating Border Radii (Squircle)...")
        
        violations = []
        for css_file in self.css_files:
            if not css_file.exists():
                continue
                
            with open(css_file, 'r') as f:
                content = f.read()
            
            # Check for border-radius values
            import re
            radius_values = re.findall(r'border-radius:\s*(\d+)px', content)
            
            expected = [12, 18, 20]
            for val in radius_values:
                num = int(val)
                if num not in expected and num != 50 and num != 999:  # 50% for circles, 999px for pills
                    violations.append(f"Non-standard radius: {num}px in {css_file.name}")
        
        if violations:
            self.results.append({
                "test": "Squircle Radii",
                "status": "⚠️  WARNINGS",
                "details": violations[:5]
            })
        else:
            self.results.append({

                "status": "✅ PASSED",
                "details": []
            })
    
    def validate_touch_targets(self):
        """Validate minimum touch target sizes (40px / 2.5rem)"""
        print("👆 Validating Touch Target Sizes...")
        
        # This requires running browser inspection
        # For now, check CSS declarations
        violations = []
        
        for css_file in self.css_files:

            # Check for small widths/heights on interactive elements
            patterns = [
                (r'\.action-icon[^}]*width:\s*(\d+\.?\d*)rem', 'action-icon width'),
                (r'\.icon-btn[^}]*min-width:\s*(\d+\.?\d*)rem', 'icon-btn width'),
            ]
            
            for pattern, name in patterns:
                matches = re.findall(pattern, content, re.DOTALL)
                for match in matches:
                    size_rem = float(match)
                    if size_rem < 2.5:  # 2.5rem = 40px
                        violations.append(f"{name} too small: {size_rem}rem (min: 2.5rem)")
        
        if violations:
            self.results.append({
                "test": "Touch Target Sizes",
                "status": "❌ FAILED",
                "details": violations
            })
        else:

    def validate_motion(self):

        print("🎬 Validating Motion Design...")
        
        violations = []
        correct_count = 0
        
        for css_file in self.css_files:
            if not css_file.exists():
                continue
                
            with open(css_file, 'r') as f:
                content = f.read()
            
            # Check for Apple spring easing
            if "cubic-bezier(0.16, 1, 0.3, 1)" in content:
                correct_count += 1
            
            # Check for linear transitions (should be avoided)
            import re
            if re.search(r'transition:.*\blinear\b', content):
                violations.append("Linear easing detected (use spring instead)")
        
        if violations:
            self.results.append({
                "test": "Motion Design",
                "status": "⚠️  WARNINGS",
                "details": violations
            })
        elif correct_count > 0:
            self.results.append({
                "test": "Motion Design",
                "status": "✅ PASSED",
                "details": [f"Apple spring easing found in {correct_count} file(s)"]
            })
        else:

                "test": "Motion Design",
                "status": "⚠️  WARNING",
                "details": ["Spring easing not detected"]
            })

#!/usr/bin/env python3
"""
Update README.md with entertainment library statistics and categorized content.
"""
import json
from pathlib import Path
from collections import defaultdict


def load_library(json_path):
    """Load the library data from JSON file."""
    with open(json_path, 'r') as f:
        return json.load(f)


def calculate_stats(library):
    """Calculate statistics from the library."""
    stats = {
        'total': len(library),
        'watched': sum(1 for item in library if item['status'] == 'Watched'),
        'watching': sum(1 for item in library if item['status'] == 'Watching'),
        'plan_to_watch': sum(1 for item in library if item['status'] == 'Plan to Watch'),
        'recommended': sum(1 for item in library if item['status'] == 'Recommended'),
        'movies': sum(1 for item in library if item['type'] == 'Movie'),
        'series': sum(1 for item in library if item['type'] == 'Series'),
    }
    
    # Calculate average rating for watched items
    watched_ratings = [item['rating'] for item in library if item['status'] == 'Watched' and item['rating'] > 0]
    stats['avg_rating'] = round(sum(watched_ratings) / len(watched_ratings), 1) if watched_ratings else 0
    
    return stats


def format_item_section(item):
    """Format an item for display in README."""
    lines = []
    tags_str = ", ".join(item['tags']) if item['tags'] else ""
    lines.append(f"### {item['title']}")
    lines.append(f"- **Type**: {item['type']}")
    if tags_str:
        lines.append(f"- **Tags**: {tags_str}")
    lines.append("")
    return lines


def categorize_by_status(library):
    """Categorize library items by status."""
    categories = defaultdict(list)
    for item in library:
        categories[item['status']].append(item)
    return categories


def categorize_by_type(library):
    """Categorize library items by type."""
    categories = defaultdict(list)
    for item in library:
        categories[item['type']].append(item)
    return categories


def generate_readme(library, stats):
    """Generate the README.md content."""
    readme = []
    
    # Header
    readme.append("# Entertainment Tracker")
    readme.append("")
    readme.append("My personal entertainment library - tracking movies and series I've watched or plan to watch.")
    readme.append("")
    readme.append("**My Tastes**: High-concept Sci-Fi, Gritty Noir, & Indian Regional Gems")
    readme.append("")
    
    # Statistics
    readme.append("## 📊 Statistics")
    readme.append("")
    readme.append(f"- **Total Items**: {stats['total']}")
    readme.append(f"- **Total Watched**: {stats['watched']}")
    readme.append(f"- **Currently Watching**: {stats['watching']}")
    readme.append(f"- **Plan to Watch**: {stats['plan_to_watch']}")
    readme.append(f"- **Recommended**: {stats['recommended']}")
    readme.append(f"- **Movies**: {stats['movies']}")
    readme.append(f"- **Series**: {stats['series']}")
    if stats['avg_rating'] > 0:
        readme.append(f"- **Average Rating**: {stats['avg_rating']}/10")
    readme.append("")
    
    # Categorize by status
    categories = categorize_by_status(library)
    
    # Watched section
    if 'Watched' in categories:
        readme.append("## ✅ Watched")
        readme.append("")
        for item in sorted(categories['Watched'], key=lambda x: x['rating'], reverse=True):
            tags_str = ", ".join(item['tags']) if item['tags'] else ""
            rating_str = f"⭐ {item['rating']}/10" if item['rating'] > 0 else ""
            readme.append(f"### {item['title']}")
            readme.append(f"- **Type**: {item['type']}")
            if rating_str:
                readme.append(f"- **Rating**: {rating_str}")
            if tags_str:
                readme.append(f"- **Tags**: {tags_str}")
            readme.append("")
    
    # Currently Watching section
    if 'Watching' in categories:
        readme.append("## 🎬 Currently Watching")
        readme.append("")
        for item in categories['Watching']:
            readme.extend(format_item_section(item))
    
    # Recommended section
    if 'Recommended' in categories:
        readme.append("## 🌟 Recommended")
        readme.append("")
        for item in categories['Recommended']:
            readme.extend(format_item_section(item))
    
    # Plan to Watch section
    if 'Plan to Watch' in categories:
        readme.append("## 📝 Plan to Watch")
        readme.append("")
        for item in categories['Plan to Watch']:
            readme.extend(format_item_section(item))
    
    # Footer
    readme.append("---")
    readme.append("")
    readme.append("*This README is automatically generated from `data/library.json`*")
    readme.append("")
    
    return "\n".join(readme)


def main():
    """Main function to update README."""
    # Get paths
    base_path = Path(__file__).parent.parent
    json_path = base_path / 'data' / 'library.json'
    readme_path = base_path / 'README.md'
    
    # Load library
    library = load_library(json_path)
    
    # Calculate statistics
    stats = calculate_stats(library)
    
    # Generate README content
    readme_content = generate_readme(library, stats)
    
    # Write README
    with open(readme_path, 'w') as f:
        f.write(readme_content)
    
    print(f"✅ README.md updated successfully!")
    print(f"   Total items: {stats['total']}")
    print(f"   Watched: {stats['watched']}")


if __name__ == "__main__":
